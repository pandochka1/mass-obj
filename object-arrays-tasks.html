10 задач с пошаговым разбором

В каждой задаче данные запрашиваются у пользователя через prompt(), а результат выводится в консоль, причём строки консоли собраны через шаблонные строки — обратные кавычки `` и подстановку ${}. Ниже — работающая интерактивная версия и полный код, который можно скопировать и запустить в Node.js. Обратите внимание: prompt() работает в браузере, а в Node.js чтение ввода выполняет readline.

Задача 1

Создание массива из N записей и вывод

Запросить N (количество пользователей). N раз запросить имя и баллы, собрать массив объектов, вывести каждую запись через шаблонную строку.
Как это работает: цикл по N итераций, на каждой — prompt для имени и баллов, затем push объекта. Вывод — цикл с console.log(`Имя — баллы`). 
let n = +prompt("Сколько пользователей?");
let users = [];

for (let i = 0; i < n; i++) {
  let name = prompt("Имя пользователя:");
  let score = +prompt("Баллы:");
  users.push({ name: name, score: score });
}

for (let i = 0; i < users.length; i++) {
  console.log(`${users[i].name} — ${users[i].score} баллов`);
}
Построчное пояснение
let n = +prompt("Сколько пользователей?");Запрашиваем количество. Унарный + превращает строку из prompt в число.
let users = [];Пустой массив для накопления записей.
for (let i = 0; i < n; i++)Повторяем ровно n раз — столько записей создадим.
let name = prompt("Имя...");Запрашиваем имя очередного пользователя.
let score = +prompt("Баллы:");Запрашиваем баллы и сразу преобразуем в число.
users.push({ name, score });Добавляем объект с полями name и score в конец массива.
for (... i < users.length ...)Второй цикл — вывод всех созданных записей.
console.log(`...${users[i].name} — ${users[i].score}...`);Шаблонная строка: подставляет значения полей в текст. ${} вставляет значение выражения.
Задача 2

Поиск записи по имени

Запросить имя и найти запись с таким именем в массиве. Вывести данные найденного пользователя через шаблонную строку. 
let users = [
  { name: "Олег", role: "user" }, { name: "Мария", role: "admin" },
  { name: "Пётр", role: "user" }
];

let target = prompt("Кого ищем?");
let found = -1;

for (let i = 0; i < users.length; i++) {
  if (users[i].name === target) { found = i; break; }
}

if (found !== -1) {
  console.log(`Найдено: ${users[found].name}, роль ${users[found].role}`);
} else {
  console.log(`Пользователь "${target}" не найден`);
}
Построчное пояснение
let target = prompt("Кого ищем?");Запрашиваем имя для поиска.
let found = -1;Индекс найденной записи; -1 — «не найдено».
if (users[i].name === target)Сравниваем поле name текущей записи с введённым именем.
found = i; break;Запоминаем индекс и прерываем цикл (первое совпадение).
if (found !== -1) {...}Проверяем результат поиска и выводим данные через шаблонную строку, либо сообщение об отсутствии.

Задача 4

Сортировка по полю

Отсортировать массив объектов по полю revenue (по возрастанию) сортировкой пузырьком. Вывести результат через шаблонную строку. 
let brands = [
  { name: "B", revenue: 200000 }, { name: "A", revenue: 150000 },
  { name: "D", revenue: 320000 }, { name: "C", revenue: 80000 }
];

for (let i = 0; i < brands.length - 1; i++) {
  for (let j = 0; j < brands.length - 1 - i; j++) {
    if (brands[j].revenue > brands[j + 1].revenue) {
      let temp = brands[j];
      brands[j] = brands[j + 1];
      brands[j + 1] = temp;
    }
  }
}

for (let i = 0; i < brands.length; i++) {
  console.log(`${brands[i].name} — ${brands[i].revenue}`);
}
Построчное пояснение
for (let i = 0; i < brands.length - 1; i++)Внешний цикл — число проходов пузырьковой сортировки.
for (let j = 0; j < brands.length - 1 - i; j++)Внутренний цикл — попарное сравнение соседних записей.
if (brands[j].revenue > brands[j + 1].revenue)Сравниваем поля revenue двух соседних записей.
let temp = brands[j]; ...Обмен целых объектов местами через временную переменную.
console.log(\`${brands[i].name} — ${brands[i].revenue}\`)Вывод отсортированного списка шаблонной строкой.
Задача 5

Сумма и среднее по полю

Найти сумму и среднее арифметическое поля revenue по всем записям. Результат вывести шаблонной строкой. 
let brands = [
  { name: "TyresPro", revenue: 245000 }, { name: "SecureCart", revenue: 312000 },
  { name: "OldShop", revenue: 98000 }, { name: "DataWare", revenue: 156000 }
];

let sum = 0;

for (let i = 0; i < brands.length; i++) {
  sum += brands[i].revenue;
}

let avg = sum / brands.length;
console.log(`Суммарная выручка: ${sum}`);
console.log(`Средняя выручка: ${Math.round(avg)}`);
Построчное пояснение
let sum = 0;Накопитель суммы.
sum += brands[i].revenue;Прибавляем поле revenue текущей записи.
let avg = sum / brands.length;Среднее = сумма / число записей.
console.log(\`Средняя выручка: ${Math.round(avg)}\`)Подстановка округлённого значения прямо в шаблонную строку.

Задача 7

Подсчёт записей по условию поля

Посчитать, сколько записей имеют поле active === true. Вывести количество через шаблонную строку. 
let brands = [
  { name: "TyresPro", revenue: 245000, active: true },
  { name: "SecureCart", revenue: 312000, active: true },
  { name: "OldShop", revenue: 98000, active: false },
  { name: "DataWare", revenue: 156000, active: true }
];

let activeCount = 0;

for (let i = 0; i < brands.length; i++) {
  if (brands[i].active) {
    activeCount++;
  }
}

console.log(`Активных брендов: ${activeCount} из ${brands.length}`);
Построчное пояснение
if (brands[i].active)Поле active — булево, поэтому можно проверять напрямую без сравнения с true (оно и так true/false).
activeCount++;Увеличиваем счётчик при «активной» записи.
console.log(\`Активных брендов: ${activeCount} из ${brands.length}\`)Вывод двух подстановок в одной шаблонной строке.
Задача 8

Подсчёт по введённой роли

Запросить роль и посчитать, сколько пользователей имеют эту роль. Результат вывести шаблонной строкой.

console.log(\`...${role}"... ${count}\`)Подстановка и роли, и количества в одну шаблонную строку.
Задача 9

Обновление поля всех записей

Добавить одинаковый бонус к полю score всех записей. Ввод бонуса через prompt, вывод — шаблонной строкой. 
let users = [
  { name: "Олег", score: 85 }, { name: "Мария", score: 92 }, { name: "Пётр", score: 78 }
];

let bonus = +prompt("Сколько добавить баллов?");

for (let i = 0; i < users.length; i++) {
  users[i].score += bonus;
}

for (let i = 0; i < users.length; i++) {
  console.log(`Обновлённый балл: ${users[i].name} → ${users[i].score}`);
}
Построчное пояснение

users[i].score += bonus;Прибавляем бонус к полю score каждой записи (мутация объекта).
console.log(\`Обновлённый балл: ${users[i].name} → ${users[i].score}\`)Вывод имени и нового значения через подстановку.
Задача 10

Отбор записей выше порога

Запросить порог и оставить в новом массиве только записи, чей score не ниже порога. Вывести оставшиеся через шаблонную строку. 
let users = [
  { name: "Олег", score: 85 }, { name: "Мария", score: 92 },
  { name: "Пётр", score: 40 }, { name: "Анна", score: 60 }
];

let threshold = +prompt("Минимальный балл:");
let kept = [];

for (let i = 0; i < users.length; i++) {

for (let i = 0; i < kept.length; i++) {
  console.log(`${kept[i].name} — ${kept[i].score}`);
}
Построчное пояснение

let kept = [];Новый массив для прошедших порог записей.
if (users[i].score >= threshold)Условие отбора по полю score.
kept.push(users[i]);Добавляем подходящую запись.
console.log(\`Прошли порог: ${kept.length} человек\`)Вывод количества отобранных записей.
console.log(\`${kept[i].name} — ${kept[i].score}\`)Вывод каждой отобранной записи. Исходный массив не изменён.

Как запускать код

В браузере: код с prompt() запускается в консоли разработчика (F12).
В Node.js: prompt() отсутствует — для ввода используйте readline или заранее заданные данные, а вывод через console.log работает как обычно.
Способ запуска: node имя_файла.js

Шаблонные строки (`` и ${}) работают в обоих окружениях без настроек.
